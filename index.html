async function sendAI(message, mode = "analyze") { try { const symbol = currentSymbol || "BTCUSDT"; /* * Get the currently selected timeframe. * Change this selector if your existing chart uses another ID. */ const timeframe = document.querySelector(".timeframe.active")?.dataset?.interval || window.currentInterval || "15m"; /* * Ask Binance for REAL candle data. */ const binanceInterval = timeframe; const response = await fetch( `https://api.binance.com/api/v3/klines?symbol=${symbol}&interval=${binanceInterval}&limit=300` ); if (!response.ok) { throw new Error("Binance candle request failed"); } const raw = await response.json(); /* * Convert Binance candles into clean OHLCV data. */ const candles = raw.map((c, i) => ({ index: i, time: Number(c[0]), open: Number(c[1]), high: Number(c[2]), low: Number(c[3]), close: Number(c[4]), volume: Number(c[5]) })); /* * Send REAL chart data to our AI backend. */ const aiResponse = await fetch("/api/ai", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ mode: mode, message: message, symbol: symbol, timeframe: binanceInterval, candles: candles }) }); const result = await aiResponse.json(); if (!aiResponse.ok) { throw new Error( result.error || "AI request failed" ); } /* * Display AI answer. */ const answer = result.answer || result.response || "AI returned no answer."; /* * Use your existing AI output element here. */ const aiOutput = document.getElementById("aiResponse") || document.getElementById("ai-output") || document.getElementById("aiResult"); if (aiOutput) { aiOutput.innerText = answer; } return answer; } catch (error) { console.error("AI ERROR:", error); const aiOutput = document.getElementById("aiResponse") || document.getElementById("ai-output") || document.getElementById("aiResult"); if (aiOutput) { aiOutput.innerText = "AI connection error: " + error.message; } return null; } }
