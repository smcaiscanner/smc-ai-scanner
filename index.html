export async function onRequestPost(context) { try { const body = await context.request.json(); const message = String( body.message || "" ).trim(); const mode = String( body.mode || "chat" ); const symbol = String( body.symbol || "BINANCE:BTCUSDT" ); const timeframe = String( body.timeframe || "15m" ); const history = Array.isArray(body.history) ? body.history.slice(-10) : []; if (!message) { return Response.json( { error: "Message is required." }, { status: 400 } ); } /* * Check Cloudflare AI binding */ if (!context.env.AI) { return Response.json( { error: "Workers AI binding is missing. " + "Create a Workers AI binding named AI " + "in your Cloudflare Pages project." }, { status: 500 } ); } /* * SMC SYSTEM PROMPT */ const systemPrompt = ` You are SMC AI inside the SMC AI Scanner crypto charting website. Your job is to help the user learn and apply Smart Money Concepts. Focus on: Buy-side liquidity (BSL) Sell-side liquidity (SSL) Liquidity sweeps / SFP Equal highs Equal lows Major swing liquidity Previous day high / low Previous week high / low Market Structure Shift (MSS) CHoCH BOS Displacement Fair Value Gap (FVG) Inverse FVG Order Blocks Premium / Discount Multi-timeframe confirmation USER'S SMC METHODOLOGY: The user wants higher-timeframe liquidity confirmation first. Then lower-timeframe structure confirmation. Then retracement into FVG or Order Block. The AI should help convert the user's examples into objective rules that can later be implemented by a deterministic scanner. WHEN USER TEACHES A CONCEPT: Do not silently invent rules. Extract the user's rule. Give the concept a name. Convert it into objective conditions. Explain confirmation conditions. Explain invalidation conditions. Explain what OHLCV/chart data is required. Create a concise scanner rule. WHEN ANALYZING: Never claim to see chart candles unless candle data was actually provided. Never invent prices. Never invent whale activity. Never invent a signal. If data is missing, tell the user exactly what data is needed. This is an analysis/scanning tool and not financial advice. Current symbol: ${symbol} Current timeframe: ${timeframe} Current mode: ${mode} `; /* * BUILD MESSAGE HISTORY */ const messages = [ { role: "system", content: systemPrompt } ]; for ( const item of history ) { if ( item && ( item.role === "user" || item.role === "assistant" ) && typeof item.content === "string" ) { messages.push({ role: item.role, content: item.content }); } } /* * CURRENT USER MESSAGE */ messages.push({ role: "user", content: message }); /* * CLOUDFLARE WORKERS AI */ const result = await context.env.AI.run( "@cf/meta/llama-3.1-8b-instruct", { messages: messages, max_tokens: 900, temperature: 0.2 } ); /* * GET RESPONSE */ let answer = ""; if ( typeof result === "string" ) { answer = result; } else if ( result && typeof result.response === "string" ) { answer = result.response; } else if ( result && result.result && typeof result.result.response === "string" ) { answer = result.result.response; } else { answer = JSON.stringify( result ); } /* * RETURN AI RESPONSE */ return Response.json({ answer: answer }); } catch (error) { return Response.json( { error: error && error.message ? error.message : "Workers AI request failed." }, { status: 500 } ); } }
